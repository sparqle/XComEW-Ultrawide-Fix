//
//     ScaleFactor = ScreenHeight / 720.0
//     ScreenX = ScreenWidth * 0.5 + (UIX - 640.0) * ScaleFactor
//     ScreenY = UIY * ScaleFactor
//
// Replace the native implementation with a simulated UnrealScript function
// and add the controller, local player, scene view, and scale locals it needs.
// If the controller, player, or scene view is unavailable, return the input point.
//
// Intended result:
//   - correct UI-to-screen coordinate mapping on ultrawide
//   - no horizontal stretching of the virtual UI coordinate space
//   - 16:9 behavior remains unchanged
//
// Apply once to the original native function. Do not also apply the standalone
// ConvertUIPointToScreenCoordinate patch to the same package.
// ------------------------------------------------------------

// ------------------------------------------------------------
// Center modal dialogue box on ultrawide resolutions.
//
// The DialogueBox.realise actionscript function does:
//
//  this.bg._x = Stage.width * 0.5;
//  this.bg._y = Stage.height * 0.5;
//
// which is wrong for ultrawide resolutions as the coordinate system is 1280x720 centered,
// with negative values for x and y if wider.
//
// The fix is to replace the above with a static to the known coordinate system center:
//
//  this.bg._x = 1280 * 0.5 - 720 * 0.5;
//  this.bg._y = 1280 * 0.5 - 720 * 0.5;
//
// Other UI elements use anchors which properly take the resolution into account.
// ------------------------------------------------------------

UPK_FILE=UICollection_Common_SF.upk
OBJECT=gfxDialogueBox.DialogueBox
